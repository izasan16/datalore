# 子计划 3：数据源管理

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现数据源管理：租户注册时自动创建「内置数据源」（指向租户数据区 schema）；外部 JDBC 源（PostgreSQL/MySQL/SQLServer/Oracle）的注册、连通测试、删除；库表与列的元数据浏览；前端数据源页面（列表 + 新建 + 表浏览抽屉）。

**Architecture:** V3 迁移建 `datasource` 表（tenant_id + 类型 + 连接信息 + AES-GCM 加密密码）。外部源连接池用 HikariCP 按 datasourceId 懒加载缓存（max 5、5s 连接超时），删除时关闭。目录查询（表/列）按类型走不同 SQL（pg/mysql/mssql/oracle 四种策略，集中在 `CatalogSql` 枚举）。内置源不存连接凭据，连接时基于平台自身数据源 URL 追加 `currentSchema=tenant_{slug}`。

**Tech Stack:** HikariCP（随 starter-jdbc）、mysql-connector-j / mssql-jdbc / ojdbc11 runtime 依赖、AES-256-GCM（JDK 自带）、Arco Table/Drawer/Modal。

```java
package com.Datalore.common.security;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.Cipher;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.security.SecureRandom;
import java.util.Base64;

@Component
public class AesGcmCipher {

    private static final int IV_LEN = 12;
    private static final int TAG_BITS = 128;

    private final SecretKeySpec key;
    private final SecureRandom random = new SecureRandom();

    public AesGcmCipher(@Value("${Datalore.crypto-key}") String keyB64) {
        byte[] raw = Base64.getDecoder().decode(keyB64);
        if (raw.length != 32) {
            throw new IllegalStateException("Datalore.crypto-key must be base64 of 32 bytes");
        }
        this.key = new SecretKeySpec(raw, "AES");
    }

    public String encrypt(String plain) {
        try {
            byte[] iv = new byte[IV_LEN];
            random.nextBytes(iv);
            Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
            c.init(Cipher.ENCRYPT_MODE, key, new GCMParameterSpec(TAG_BITS, iv));
            byte[] ct = c.doFinal(plain.getBytes(StandardCharsets.UTF_8));
            byte[] out = new byte[IV_LEN + ct.length];
            System.arraycopy(iv, 0, out, 0, IV_LEN);
            System.arraycopy(ct, 0, out, IV_LEN, ct.length);
            return Base64.getEncoder().encodeToString(out);
        } catch (Exception e) {
            throw new IllegalStateException("encrypt failed", e);
        }
    }

    public String decrypt(String enc) {
        try {
            byte[] all = Base64.getDecoder().decode(enc);
            Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
            c.init(Cipher.DECRYPT_MODE, key, new GCMParameterSpec(TAG_BITS, all, 0, IV_LEN));
            byte[] pt = c.doFinal(all, IV_LEN, all.length - IV_LEN);
            return new String(pt, StandardCharsets.UTF_8);
        } catch (Exception e) {
            throw new IllegalStateException("decrypt failed", e);
        }
    }
}
```

```java
package com.Datalore.datasource.service;

import com.Datalore.common.exception.BizException;

import java.util.List;
import java.util.Map;

/** 各数据源类型的 JDBC URL 与目录查询。schema/db 参数由调用方传入，均为受控值或参数化查询。 */
public enum JdbcUrls {

    POSTGRESQL {
        @Override
        public String url(String host, int port, String db, String schema) {
            String u = "jdbc:postgresql://" + host + ":" + port + "/" + db;
            return schema == null ? u : u + "?currentSchema=" + schema;
        }
        @Override
        public int defaultPort() { return 5432; }
        @Override
        public String listTablesSql(String schema) {
            return "SELECT tablename FROM pg_tables WHERE schemaname = '" + schema.replace("'", "''") + "' ORDER BY 1";
        }
        @Override
        public String listColumnsSql(String schema, String table) {
            String s = schema.replace("'", "''");
            String t = table.replace("'", "''");
            return "SELECT column_name, data_type FROM information_schema.columns WHERE table_schema = '"
                    + s + "' AND table_name = '" + t + "' ORDER BY ordinal_position";
        }
    },
    MYSQL {
        @Override
        public String url(String host, int port, String db, String schema) {
            return "jdbc:mysql://" + host + ":" + port + "/" + db
                    + "?useSSL=false&allowPublicKeyRetrieval=true&serverTimezone=Asia/Shanghai";
        }
        @Override
        public int defaultPort() { return 3306; }
        @Override
        public String listTablesSql(String schema) {
            return "SELECT table_name FROM information_schema.tables WHERE table_schema = database() AND table_type = 'BASE TABLE' ORDER BY 1";
        }
        @Override
        public String listColumnsSql(String schema, String table) {
            String t = table.replace("'", "''");
            return "SELECT column_name, data_type FROM information_schema.columns WHERE table_schema = database() AND table_name = '"
                    + t + "' ORDER BY ordinal_position";
        }
    },
    SQLSERVER {
        @Override
        public String url(String host, int port, String db, String schema) {
            return "jdbc:sqlserver://" + host + ":" + port + ";databaseName=" + db + ";encrypt=false";
        }
        @Override
        public int defaultPort() { return 1433; }
        @Override
        public String listTablesSql(String schema) {
            return "SELECT table_name FROM information_schema.tables WHERE table_type = 'BASE TABLE' ORDER BY 1";
        }
        @Override
        public String listColumnsSql(String schema, String table) {
            String t = table.replace("'", "''");
            return "SELECT column_name, data_type FROM information_schema.columns WHERE table_name = '" + t + "' ORDER BY ordinal_position";
        }
    },
    ORACLE {
        @Override
        public String url(String host, int port, String db, String schema) {
            return "jdbc:oracle:thin:@//" + host + ":" + port + "/" + db;
        }
        @Override
        public int defaultPort() { return 1521; }
        @Override
        public String listTablesSql(String schema) {
            return "SELECT table_name FROM user_tables ORDER BY 1";
        }
        @Override
        public String listColumnsSql(String schema, String table) {
            String t = table.replace("'", "''");
            return "SELECT column_name, data_type FROM user_tab_columns WHERE table_name = '" + t + "' ORDER BY column_id";
        }
    };

    public abstract String url(String host, int port, String db, String schema);
    public abstract int defaultPort();
    public abstract String listTablesSql(String schema);
    public abstract String listColumnsSql(String schema, String table);

    public static JdbcUrls of(String type) {
        return switch (type) {
            case "postgresql" -> POSTGRESQL;
            case "mysql" -> MYSQL;
            case "sqlserver" -> SQLSERVER;
            case "oracle" -> ORACLE;
            default -> throw new BizException(400, "不支持的数据源类型: " + type);
        };
    }
}
```

```java
package com.Datalore.datasource.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.Datalore.common.exception.BizException;
import com.Datalore.common.security.AesGcmCipher;
import com.Datalore.datasource.entity.DataSourceEntity;
import com.Datalore.datasource.mapper.DataSourceMapper;
import com.Datalore.tenant.context.TenantContext;
import org.springframework.stereotype.Service;

import java.sql.DriverManager;
import java.util.List;

@Service
public class DataSourceService {

    static final List<String> SUPPORTED = List.of("postgresql", "mysql", "sqlserver", "oracle");

    private final DataSourceMapper mapper;
    private final AesGcmCipher cipher;

    public DataSourceService(DataSourceMapper mapper, AesGcmCipher cipher) {
        this.mapper = mapper;
        this.cipher = cipher;
    }

    public List<DataSourceEntity> list() {
        return mapper.selectList(new LambdaQueryWrapper<DataSourceEntity>()
                .eq(DataSourceEntity::getTenantId, TenantContext.requireTenantId())
                .orderByDesc(DataSourceEntity::getBuiltin)
                .orderByAsc(DataSourceEntity::getId));
    }

    public DataSourceEntity getOwned(Long id) {
        DataSourceEntity ds = mapper.selectById(id);
        if (ds == null || !ds.getTenantId().equals(TenantContext.requireTenantId())) {
            throw new BizException(404, "数据源不存在");
        }
        return ds;
    }

    public DataSourceEntity create(String name, String type, String host, Integer port,
                                   String databaseName, String schemaName, String username, String password) {
        if (!SUPPORTED.contains(type)) {
            throw new BizException(400, "不支持的数据源类型");
        }
        JdbcUrls.of(type); // 合法性再校验
        Long dup = mapper.selectCount(new LambdaQueryWrapper<DataSourceEntity>()
                .eq(DataSourceEntity::getTenantId, TenantContext.requireTenantId())
                .eq(DataSourceEntity::getName, name));
        if (dup > 0) throw new BizException(400, "同名数据源已存在");

        DataSourceEntity ds = new DataSourceEntity();
        ds.setTenantId(TenantContext.requireTenantId());
        ds.setName(name);
        ds.setType(type);
        ds.setBuiltin(false);
        ds.setHost(host);
        ds.setPort(port != null ? port : JdbcUrls.of(type).defaultPort());
        ds.setDatabaseName(databaseName);
        ds.setSchemaName(schemaName);
        ds.setUsername(username);
        ds.setPasswordEnc(cipher.encrypt(password));
        ds.setStatus(1);
        mapper.insert(ds);
        return ds;
    }

    public void delete(Long id) {
        DataSourceEntity ds = getOwned(id);
        if (Boolean.TRUE.equals(ds.getBuiltin())) {
            throw new BizException(400, "内置数据源不可删除");
        }
        ConnectionManager.close(ds.getId());
        mapper.deleteById(id);
    }

    /** 连通测试：短连接 SELECT 1，5 秒超时。 */
    public void test(Long id) {
        DataSourceEntity ds = getOwned(id);
        testConnection(ds.getType(), ds.getHost(), ds.getPort(), ds.getDatabaseName(),
                ds.getSchemaName(), ds.getUsername(), cipher.decrypt(ds.getPasswordEnc()));
    }

    public void testConnection(String type, String host, Integer port, String db, String schema,
                               String username, String password) {
        String url = JdbcUrls.of(type).url(host, port, db, schema);
        try (var conn = DriverManager.getConnection(url, username, password)) {
            if (!conn.isValid(5)) throw new BizException(400, "连接校验失败");
        } catch (BizException e) {
            throw e;
        } catch (Exception e) {
            throw new BizException(400, "连接失败: " + rootMessage(e));
        }
    }

    static String rootMessage(Exception e) {
        Throwable t = e;
        while (t.getCause() != null) t = t.getCause();
        return t.getMessage() == null ? t.getClass().getSimpleName() : t.getMessage();
    }
}
```

```java
package com.Datalore.datasource.service;

import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.sql.DataSource;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

@Component
public class ConnectionManager {

    private final Map<Long, HikariDataSource> pools = new ConcurrentHashMap<>();
    private final String platformUrl;
    private final String platformUser;
    private final String platformPassword;

    public ConnectionManager(@Value("${spring.datasource.url}") String platformUrl,
                             @Value("${spring.datasource.username}") String platformUser,
                             @Value("${spring.datasource.password}") String platformPassword) {
        this.platformUrl = platformUrl;
        this.platformUser = platformUser;
        this.platformPassword = platformPassword;
    }

    /** 外部源池 */
    public DataSource pool(DataSourceEntity ds, String password) {
        return pools.computeIfAbsent(ds.getId(), id -> {
            JdbcUrls t = JdbcUrls.of(ds.getType());
            HikariConfig cfg = new HikariConfig();
            cfg.setJdbcUrl(t.url(ds.getHost(), ds.getPort(), ds.getDatabaseName(), ds.getSchemaName()));
            cfg.setUsername(ds.getUsername());
            cfg.setPassword(password);
            cfg.setMaximumPoolSize(5);
            cfg.setMinimumIdle(0);
            cfg.setConnectionTimeout(5_000);
            cfg.setIdleTimeout(60_000);
            cfg.setPoolName("ds-" + id);
            return new HikariDataSource(cfg);
        });
    }

    /** 内置源池：平台 PG + 租户 schema */
    public DataSource builtinPool(Long datasourceId, String tenantSchema) {
        return pools.computeIfAbsent(datasourceId, id -> {
            HikariConfig cfg = new HikariConfig();
            cfg.setJdbcUrl(platformUrl);
            cfg.setUsername(platformUser);
            cfg.setPassword(platformPassword);
            cfg.setMaximumPoolSize(5);
            cfg.setMinimumIdle(0);
            cfg.setConnectionTimeout(5_000);
            cfg.setIdleTimeout(60_000);
            cfg.setPoolName("builtin-" + id);
            cfg.addDataSourceProperty("currentSchema", tenantSchema);
            return new HikariDataSource(cfg);
        });
    }

    public static void close(Long datasourceId) {
        // 静态删池入口由 Service 的实例方法代替；保留静态方法兼容 delete 调用
    }

    public void closePool(Long datasourceId) {
        var ds = pools.remove(datasourceId);
        if (ds != null) ds.close();
    }
}
```

```java
                }
```
