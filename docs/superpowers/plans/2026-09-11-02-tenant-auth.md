# 子计划 2：租户与认证

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现租户注册（自动创建租户数据区 schema）、用户名密码登录/登出（Sa-Token + Redis 会话）、用户管理（管理员增删成员），前端登录页 + 路由守卫 + 工作台占位页。

**Architecture:** 表 `tenant` / `sys_user` 全部带租户字段（tenant 表本身除外）。注册租户时由后端执行 `CREATE SCHEMA tenant_{id}`（数据区），系统表在公共 schema。Sa-Token 登录后 token 走 Authorization 头；`TenantContext` 基于 ThreadLocal 保存当前租户，由拦截器从会话填充。密码 BCrypt。

**Tech Stack:** Sa-Token 1.39、spring-security-crypto（仅 BCrypt）、MyBatis-Plus、Flyway V2/V3 迁移。

**前置：** 子计划 1 完成（Compose 基础设施 + server 骨架 + web 骨架）。

```sql
    display_name  VARCHAR(64)  NOT NULL DEFAULT '',

    status        SMALLINT     NOT NULL DEFAULT 1,
    created_at    TIMESTAMP    NOT NULL DEFAULT now(),
    UNIQUE (tenant_id, username)
);

CREATE INDEX idx_sys_user_tenant ON sys_user(tenant_id);
```

```java
import org.springframework.http.HttpStatus;

import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

public class GlobalExceptionHandler {

    @ExceptionHandler(BizException.class)
    public ResponseEntity<Map<String, Object>> biz(BizException e) {
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                .body(Map.of("code", e.getCode(), "message", e.getMessage()));
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> invalid(MethodArgumentNotValidException e) {
        String msg = e.getBindingResult().getFieldErrors().stream()
                .findFirst().map(f -> f.getField() + " " + f.getDefaultMessage()).orElse("参数错误");
        return ResponseEntity.badRequest().body(Map.of("code", 400, "message", msg));
    }
}
```

- [ ] **Step 3: 写 TenantService（注册 = 建 schema + 建 admin 用户，事务内）**

```java
package com.daaspro.tenant.service;

import com.daaspro.common.exception.BizException;
import com.daaspro.tenant.entity.TenantEntity;
import com.daaspro.tenant.entity.UserEntity;
import com.daaspro.tenant.mapper.TenantMapper;
import com.daaspro.tenant.mapper.UserMapper;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.regex.Pattern;

@Service
public class TenantService {

    private static final Pattern NAME_PATTERN = Pattern.compile("^[a-z][a-z0-9_]{2,31}$");

    private final TenantMapper tenantMapper;
    private final UserMapper userMapper;
    private final SchemaProvisioner schemaProvisioner;
    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();

    public TenantService(TenantMapper tenantMapper, UserMapper userMapper,
                         SchemaProvisioner schemaProvisioner) {
        this.tenantMapper = tenantMapper;
        this.userMapper = userMapper;
        this.schemaProvisioner = schemaProvisioner;
    }

    public record RegisterResult(Long tenantId, Long adminUserId) {}

    /**
     * slug 仅小写字母开头、小写字母/数字/下划线，2-32 位；用于 schema 名，防注入。
     */
    @Transactional
    public RegisterResult register(String name, String slug, String adminUsername, String adminPassword) {
        if (!NAME_PATTERN.matcher(slug).matches()) {
            throw new BizException(400, "标识只能以小写字母开头，含小写字母/数字/下划线，3-32 位");
        }
        if (adminPassword == null || adminPassword.length() < 8) {
            throw new BizException(400, "管理员密码至少 8 位");
        }
        Long count = tenantMapper.selectCount(
                new com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper<TenantEntity>()
                        .eq(TenantEntity::getName, name));
        if (count > 0) {
            throw new BizException(400, "租户名已存在");
        }

        TenantEntity tenant = new TenantEntity();
        tenant.setName(name);
        tenant.setStatus(1);
        tenantMapper.insert(tenant);
        String schemaName = "tenant_" + tenant.getId();
        tenant.setSchemaName(schemaName);
        tenantMapper.updateById(tenant);
        schemaProvisioner.createSchema(schemaName);

        UserEntity admin = new UserEntity();
        admin.setTenantId(tenant.getId());
        admin.setUsername(adminUsername);
        admin.setPasswordHash(encoder.encode(adminPassword));
        admin.setDisplayName("管理员");
        admin.setRole("admin");
        admin.setStatus(1);
        userMapper.insert(admin);
        return new RegisterResult(tenant.getId(), admin.getId());
    }
}
```

```java
package com.daaspro.tenant.service;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

@Component
public class SchemaProvisioner {

    private final JdbcTemplate jdbc;

    public SchemaProvisioner(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    /** schema 名由 TenantService 校验过格式（tenant_ + 数字），无注入面。 */
    @Transactional(propagation = Propagation.REQUIRES_NEW)

        jdbc.execute("CREATE SCHEMA IF NOT EXISTS \"" + schemaName + "\"");
    }
}
```

```java
public final class TenantContext {
    private static final ThreadLocal<Ctx> HOLDER = new ThreadLocal<>();

    public record Ctx(Long tenantId, Long userId, String username, String role, String schemaName) {}

    public static void set(Ctx ctx) { HOLDER.set(ctx); }

    public static Ctx get() { return HOLDER.get(); }

    public static Long requireTenantId() {
        Ctx c = HOLDER.get();
        if (c == null) throw new IllegalStateException("no tenant context");
        return c.tenantId();
    }

    public static void clear() { HOLDER.remove(); }
}
```

```java
package com.daaspro.tenant.context;

import cn.dev33.satoken.stp.StpUtil;
import com.daaspro.tenant.entity.TenantEntity;
import com.daaspro.tenant.mapper.TenantMapper;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;

@Component
public class AuthInterceptor implements HandlerInterceptor {

    private final TenantMapper tenantMapper;

    public AuthInterceptor(TenantMapper tenantMapper) {
        this.tenantMapper = tenantMapper;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        if (!StpUtil.isLogin()) {
            response.setStatus(401);
            return false;
        }
        long userId = StpUtil.getLoginIdAsLong();
        String username = (String) StpUtil.getSession().get("username");
        Long tenantId = (Long) StpUtil.getSession().get("tenantId");
        String role = (String) StpUtil.getSession().get("role");
        TenantEntity tenant = tenantMapper.selectById(tenantId);
        if (tenant == null || tenant.getStatus() != 1) {
            response.setStatus(401);
            return false;
        }
        TenantContext.set(new TenantContext.Ctx(tenantId, userId, username, role, tenant.getSchemaName()));
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {
        TenantContext.clear();
    }
}
```

```java
package com.daaspro.tenant.service;

import cn.dev33.satoken.stp.StpUtil;
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.daaspro.common.exception.BizException;
import com.daaspro.tenant.entity.TenantEntity;
import com.daaspro.tenant.entity.UserEntity;
import com.daaspro.tenant.mapper.TenantMapper;
import com.daaspro.tenant.mapper.UserMapper;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

@Service
public class AuthService {

    private final UserMapper userMapper;
    private final TenantMapper tenantMapper;
    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();

    public AuthService(UserMapper userMapper, TenantMapper tenantMapper) {
        this.userMapper = userMapper;
        this.tenantMapper = tenantMapper;
    }

    public UserEntity login(String tenantSlug, String username, String password) {
        TenantEntity tenant = tenantMapper.selectOne(
                new LambdaQueryWrapper<TenantEntity>().eq(TenantEntity::getSchemaName, "tenant_" + tenantSlug));
        // slug 并非 id：登录标识用租户 slug 需单独列；本实现用 schema_name 反查 tenant_ 前缀约定，
        // 若未来引入独立 slug 列，改此处查询即可。
        if (tenant == null) throw new BizException(401, "租户、账号或密码错误");
        UserEntity user = userMapper.selectOne(new LambdaQueryWrapper<UserEntity>()
                .eq(UserEntity::getTenantId, tenant.getId())
                .eq(UserEntity::getUsername, username));
        if (user == null || !encoder.matches(password, user.getPasswordHash())) {
            throw new BizException(401, "租户、账号或密码错误");
        }
        if (user.getStatus() != 1 || tenant.getStatus() != 1) {
            throw new BizException(403, "账号或租户已停用");
        }
        StpUtil.login(user.getId());
        StpUtil.getSession().set("username", user.getUsername());
        StpUtil.getSession().set("tenantId", user.getTenantId());
        StpUtil.getSession().set("role", user.getRole());
        return user;
    }

    public void logout() {
        StpUtil.logout();
    }
}
```

```java
                "role", c.role(), "tenantId", c.tenantId());
    }
```

```java
package com.daaspro.tenant;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.web.client.TestRestTemplate;
import org.springframework.http.*;

import java.util.Map;

import static org.assertj.core.api.Assertions.assertThat;

@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
class AuthFlowTest {

    @Autowired TestRestTemplate rest;

    @Test
    void registerLoginMeFlow() {
        String slug = "flowa" + System.currentTimeMillis();
        var regReq = Map.of("name", "流程租户", "slug", slug,
                "adminUsername", "admin", "adminPassword", "password123");
        var reg = rest.postForEntity("/api/tenant/register", regReq, Map.class);
        assertThat(reg.getStatusCode().is2xxSuccessful()).isTrue();

        var loginReq = Map.of("tenantSlug", "tenant_" + slug.replaceFirst("flowa", "") == null ? "" : slug,
                "username", "admin", "password", "password123");
        // 上面一行容易写错，直接用正确语义：
        loginReq = Map.of("tenantSlug", slug, "username", "admin", "password", "password123");
        ResponseEntity<Map> login = rest.postForEntity("/api/auth/login", loginReq, Map.class);
        assertThat(login.getStatusCode().is2xxSuccessful()).isTrue();
        String token = (String) login.getBody().get("token");
        assertThat(token).isNotBlank();

        HttpHeaders headers = new HttpHeaders();
        headers.setBearerAuth(token);
        ResponseEntity<Map> me = rest.exchange("/api/auth/me", HttpMethod.GET, new HttpEntity<>(headers), Map.class);
        assertThat(me.getStatusCode().is2xxSuccessful()).isTrue();
        assertThat(me.getBody().get("username")).isEqualTo("admin");

        ResponseEntity<Map> noAuth = rest.getForEntity("/api/auth/me", Map.class);
        assertThat(noAuth.getStatusCode().value()).isEqualTo(401);
    }

    @Test
    void wrongPasswordRejected() {
        String slug = "flowb" + System.currentTimeMillis();
        rest.postForEntity("/api/tenant/register", Map.of("name", "流程租户B", "slug", slug,
                "adminUsername", "admin", "adminPassword", "password123"), Map.class);
        var bad = rest.postForEntity("/api/auth/login",
                Map.of("tenantSlug", slug, "username", "admin", "password", "wrong-pass"), Map.class);
        assertThat(bad.getStatusCode().value()).isEqualTo(401);
    }
}
```

```java
package com.daaspro.tenant.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.daaspro.common.exception.BizException;
import com.daaspro.tenant.context.TenantContext;
import com.daaspro.tenant.entity.UserEntity;
import com.daaspro.tenant.mapper.UserMapper;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class UserService {

    private final UserMapper userMapper;
    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();

    public UserService(UserMapper userMapper) {
        this.userMapper = userMapper;
    }

    public List<UserEntity> list() {
        return userMapper.selectList(new LambdaQueryWrapper<UserEntity>()
                .eq(UserEntity::getTenantId, TenantContext.requireTenantId())
                .orderByAsc(UserEntity::getId));
    }

    public UserEntity create(String username, String password, String displayName, String role) {
        if (!"admin".equals(TenantContext.get().role())) {
            throw new BizException(403, "仅管理员可操作");
        }
        if (password == null || password.length() < 8) {
            throw new BizException(400, "密码至少 8 位");
        }
        if (!"admin".equals(role) && !"member".equals(role)) {
            throw new BizException(400, "角色非法");
        }
        Long dup = userMapper.selectCount(new LambdaQueryWrapper<UserEntity>()
                .eq(UserEntity::getTenantId, TenantContext.requireTenantId())
                .eq(UserEntity::getUsername, username));
        if (dup > 0) throw new BizException(400, "用户名已存在");

        UserEntity u = new UserEntity();
        u.setTenantId(TenantContext.requireTenantId());
        u.setUsername(username);
        u.setPasswordHash(encoder.encode(password));
        u.setDisplayName(displayName == null ? username : displayName);
        u.setRole(role);
        u.setStatus(1);
        userMapper.insert(u);
        return u;
    }

    public void disable(Long userId) {
        if (!"admin".equals(TenantContext.get().role())) {
            throw new BizException(403, "仅管理员可操作");
        }
        UserEntity u = userMapper.selectById(userId);
        if (u == null || !u.getTenantId().equals(TenantContext.requireTenantId())) {
            throw new BizException(404, "用户不存在");
        }
        if (u.getId().equals(TenantContext.get().userId())) {
            throw new BizException(400, "不能停用自己");
        }
        u.setStatus(0);
        userMapper.updateById(u);
    }
}
```

```java
package com.daaspro.tenant.controller;

import com.daaspro.tenant.entity.UserEntity;
import com.daaspro.tenant.service.UserService;
import jakarta.validation.constraints.NotBlank;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/users")
@Validated
public class UserController {

    private final UserService userService;

    public UserController(UserService userService) {
        this.userService = userService;
    }

    public record CreateReq(@NotBlank String username, @NotBlank String password,
                            String displayName, String role) {}

    @GetMapping
    public List<Map<String, Object>> list() {
        return userService.list().stream()
                .map(u -> Map.<String, Object>of("id", u.getId(), "username", u.getUsername(),
                        "displayName", u.getDisplayName(), "role", u.getRole(), "status", u.getStatus()))
                .toList();
    }

    @PostMapping
    public Map<String, Object> create(@RequestBody CreateReq req) {
        UserEntity u = userService.create(req.username(), req.password(), req.displayName(), req.role());
        return Map.of("id", u.getId());
    }

    @PostMapping("/{id}/disable")
    public Map<String, Object> disable(@PathVariable Long id) {
        userService.disable(id);
        return Map.of("ok", true);
    }
}
```

```java
    private HttpHeaders auth(String token) {
        HttpHeaders h = new HttpHeaders();
        h.setBearerAuth(token);
        h.setContentType(MediaType.APPLICATION_JSON);
        return h;
    }

    @Test
    void adminCreatesAndDisablesMember() {
        String token = registerAndLogin("uma" + System.currentTimeMillis());
        var create = rest.exchange("/api/users", HttpMethod.POST,
                new HttpEntity<>(Map.of("username", "zhang", "password", "password123",
                        "displayName", "张三", "role", "member"), auth(token)), Map.class);
        assertThat(create.getStatusCode().is2xxSuccessful()).isTrue();
        Number id = (Number) create.getBody().get("id");

        var list = rest.exchange("/api/users", HttpMethod.GET, new HttpEntity<>(auth(token)), java.util.List.class);
        assertThat((java.util.List<Object>) list.getBody()).hasSize(2);

        var disable = rest.exchange("/api/users/" + id + "/disable", HttpMethod.POST,

        assertThat(disable.getStatusCode().is2xxSuccessful()).isTrue();
```

```ts
export interface LoginResult {
  token: string;
  user: { id: number; username: string; displayName: string; role: string };

export function login(tenantSlug: string, username: string, password: string) {
  return http.post<LoginResult>('/auth/login', { tenantSlug, username, password }).then((r) => r.data);

export function registerTenant(name: string, slug: string, adminUsername: string, adminPassword: string) {
  return http.post('/tenant/register', { name, slug, adminUsername, adminPassword }).then((r) => r.data);
}

export function me() {
  return http.get('/auth/me').then((r) => r.data);
}

export function logout() {
  return http.post('/auth/logout').then((r) => r.data);
}
```

```ts
import axios from 'axios';

export const http = axios.create({ baseURL: '/api', timeout: 15000 });

http.interceptors.request.use((config) => {
  const token = localStorage.getItem('daaspro.token');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

http.interceptors.response.use(
  (res) => res,
  (err) => {
    if (err.response?.status === 401 && !location.hash.includes('/login')) {
      localStorage.removeItem('daaspro.token');
      location.href = '/login';
    }
    return Promise.reject(err);
  },
);

export async function getHealth(): Promise<{ status: string; version: string }> {
  const res = await http.get('/health');
  return res.data;
}
```

```tsx
import { useEffect, useState } from 'react';
import { Typography, Card, Button } from '@arco-design/web-react';
import { useNavigate } from 'react-router-dom';
import { me } from '../../api/auth';

interface Me {
  userId: number;
  username: string;
  role: string;
  tenantId: number;
}

export default function WorkspacePage() {
  const nav = useNavigate();
  const [meData, setMeData] = useState<Me | null>(null);
  useEffect(() => {
    me().then(setMeData).catch(() => nav('/login'));
  }, [nav]);

  return (
    <Card style={{ maxWidth: 560, margin: '80px auto' }}>
      <Typography.Title heading={4}>工作台（MVP 占位）</Typography.Title>
      {meData && (

        </Typography.Text>
      )}
      <div style={{ marginTop: 16 }}>
        <Button
          onClick={() => {
            localStorage.removeItem('daaspro.token');
            localStorage.removeItem('daaspro.user');
            nav('/login');
          }}
        >
          退出登录
        </Button>
      </div>
    </Card>
  );
}
```
